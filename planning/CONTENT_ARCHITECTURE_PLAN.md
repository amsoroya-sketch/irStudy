## 📱 Technical Implementation

### Database Schema

```sql
-- Sections table
CREATE TABLE sections (
    id UUID PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT,
    sort_order INTEGER,
    created_at TIMESTAMP
);

-- Chapters table
CREATE TABLE chapters (
    id UUID PRIMARY KEY,
    section_id UUID REFERENCES sections(id),
    name VARCHAR(100) NOT NULL,
    specialty VARCHAR(50),
    frequency_rating INTEGER, -- 1, 2, or 3 stars
    tier VARCHAR(20), -- 'free', 'free_sample', 'premium'
    mcq_count INTEGER DEFAULT 0,
    osce_count INTEGER DEFAULT 0,
    sort_order INTEGER
);

-- Content items
CREATE TABLE mcqs (
    id UUID PRIMARY KEY,
    chapter_id UUID REFERENCES chapters(id),
    question_text TEXT,
    difficulty VARCHAR(20),
    -- ... question fields
);

CREATE TABLE osces (
    id UUID PRIMARY KEY,
    chapter_id UUID REFERENCES chapters(id),
    scenario TEXT,
    station_type VARCHAR(50),
    -- ... OSCE fields
);

-- User subscriptions
CREATE TABLE user_subscriptions (
    id UUID PRIMARY KEY,
    user_id UUID REFERENCES users(id),
    tier VARCHAR(20),
    expires_at TIMESTAMP,
    allowed_chapters UUID[] -- For Tier 1 specialty selection
);

-- Access control
CREATE TABLE content_access (
    user_id UUID,
    chapter_id UUID,
    granted_at TIMESTAMP,
    expires_at TIMESTAMP,
    PRIMARY KEY (user_id, chapter_id)
);
```

```python
# Modular subscription system
class SubscriptionManager:
    MODULES = {
        "amc_prep": {
            "name": "AMC Preparation",
            "price_monthly": 49,
            "price_yearly": 249,
            "content_access": ["amc_mcqs", "osce_scenarios", "flashcards"],
            "features": ["adaptive_learning", "ai_tutor", "study_plans"]
        },
        "medical_school": {
            "name": "Medical School",
            "price_monthly": 29,
            "price_yearly": 149,
            "content_access": ["basic_science", "case_based", "osce_prep"],
            "features": ["university_sync", "peer_groups"]
        },
        "nursing_allied": {
            "name": "Nursing & Allied Health",
            "price_monthly": 19,
            "price_yearly": 99,
            "content_access": ["nursing_content", "drug_calc", "clinical_skills"],
            "features": ["competency_tracking"]
        },
        "clinical_search": {
            "name": "Clinical Search",
            "price_monthly": 15,
            "price_yearly": 120,
            "content_access": ["cochrane", "statpearls", "guidelines"],
            "features": ["rag_search", "cpd_tracking", "emr_integration"]
        }
    }
    
    def get_user_access(self, user_id: str) -> Dict:
        """Determine what content/features user can access"""
        subscriptions = self.get_active_subscriptions(user_id)
        
        access = {
            "content": set(),
            "features": set(),
            "rag_collections": []
        }
        
        for sub in subscriptions:
            module = self.MODULES[sub.module_id]
            access["content"].update(module["content_access"])
            access["features"].update(module["features"])
        
        # Map content to RAG collections
        if "amc_mcqs" in access["content"]:
            access["rag_collections"].extend(["medical_textbooks", "guidelines"])
        
        if "cochrane" in access["content"]:
            access["rag_collections"].append("cochrane_reviews")
        
        if "statpearls" in access["content"]:
            access["rag_collections"].append("statpearls")
        
        return access
```

### 4.2 Query Routing

```python
class RAGQueryRouter:
    """
    Route queries to appropriate RAG collections based on user type and query intent
    """
    
    def route_query(self, user: User, query: str) -> List[str]:
        """
        Determine which collections to search
        """
        collections = []
        
        # Base collection for all users
        collections.append("evidence_base")
        
        # User-type specific collections
        if user.user_type == "amc_candidate":
            collections.append("amc_clinical_knowledge")
        
        elif user.user_type == "medical_student":
            if self._is_basic_science_query(query):
                collections.append("basic_sciences")
            else:
                collections.append("amc_clinical_knowledge")
        
        elif user.user_type == "nursing_student":
            collections.append("nursing_clinical")
        
        # Query intent detection
        intent = self._detect_intent(query)
        
        if intent == "evidence_search":
            # Prioritize Cochrane
            collections.insert(0, "evidence_base")
        
        elif intent == "clinical_guideline":
            collections.append("amc_clinical_knowledge")
        
        return collections
    
    def _detect_intent(self, query: str) -> str:
        """
        Classify query intent
        """
        evidence_keywords = ["systematic review", "meta-analysis", "evidence", "cochrane"]
        guideline_keywords = ["guideline", "recommendation", "standard", "protocol"]
        clinical_keywords = ["diagnosis", "treatment", "management", "therapy"]
        
        query_lower = query.lower()
        
        if any(kw in query_lower for kw in evidence_keywords):
            return "evidence_search"
        elif any(kw in query_lower for kw in guideline_keywords):
            return "clinical_guideline"
        elif any(kw in query_lower for kw in clinical_keywords):
            return "clinical_question"
        else:
            return "general"
```

🔐 Security Implementation ▼

Vault-First Architecture

def _get_api_key(self) -> Optional[str]:
    """Retrieve Claude API key from Vault or environment."""
    try:
        # Primary Vault path
        api_key = get_vault_secret("secret/ai-osce/claude-api-key", "value")
        return api_key
    except Exception:
        try:
            # Fallback Vault path
            api_key = get_vault_secret("irStudy/claude", "api_key")
            return api_key
        except Exception:
            # Environment variable fallback (dev only)
            return None  # get_vault_secret handles env fallback

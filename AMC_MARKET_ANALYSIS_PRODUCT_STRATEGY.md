### 4.2 Adaptive Question Selection Algorithm

```python
class AdaptiveQuestionEngine:
    """
    Smart question selection: individualized vs random
    """
    
    def generate_daily_queue(
        self,
        user_id: str,
        queue_size: int = 30,
        mode: str = "adaptive"  # "adaptive" | "random" | "weakness" | "exam"
    ) -> List[MCQ]:
        """
        Generate personalized question queue based on mode
        """
        profile = self.get_user_profile(user_id)
        
        if mode == "adaptive":
            return self._adaptive_selection(profile, queue_size)
        elif mode == "random":
            return self._random_selection(profile, queue_size)
        elif mode == "weakness":
            return self._weakness_focused(profile, queue_size)
        elif mode == "exam":
            return self._exam_simulation(profile, queue_size)
    
    def _adaptive_selection(self, profile: StudentProfile, count: int) -> List[MCQ]:
        """
        Smart mix: Spaced repetition + Weakness targeting + New content
        """
        questions = []
        
        # 1. Spaced Repetition (40%)
        # Questions due for review based on SRS algorithm
        srs_due = self.srs.get_due_cards(profile.user_id, limit=int(count * 0.4))
        questions.extend(srs_due)
        
        # 2. Weakness Targeting (30%)
        # Focus on topics with <70% accuracy
        weak_topics = profile.knowledge_state["weakness_areas"][:3]
        weakness_questions = self.db.get_questions(
            topics=weak_topics,
            exclude_ids=[q.id for q in questions],
            limit=int(count * 0.3),
            difficulty="matched"  # Match user's current level
        )
        questions.extend(weakness_questions)
        
        # 3. New Content (20%)
        # Topics not yet covered or seen <3 times
        new_topics = profile.knowledge_state["unseen_topics"][:2]
        new_questions = self.db.get_questions(
            topics=new_topics,
            exclude_ids=[q.id for q in questions],
            limit=int(count * 0.2)
        )
        questions.extend(new_questions)
        
        # 4. Strength Maintenance (10%)
        # Keep strong topics fresh
        if len(questions) < count:
            strong_topics = profile.knowledge_state["strength_areas"][:1]
            maintenance = self.db.get_questions(
                topics=strong_topics,
                exclude_ids=[q.id for q in questions],
                limit=count - len(questions)
            )
            questions.extend(maintenance)
        
        # Shuffle to prevent pattern recognition
        random.shuffle(questions)
        
        return questions[:count]
    
    def _exam_simulation(self, profile: StudentProfile, count: int) -> List[MCQ]:
        """
        Mimic actual AMC exam distribution
        """
        # AMC MCQ Blueprint Distribution
        distribution = {
            "Adult Health": 0.35,
            "Women's Health": 0.15,
            "Child Health": 0.12,
            "Mental Health": 0.12,
            "Population Health": 0.08,
            "Surgery": 0.18
        }
        
        questions = []
        for category, percentage in distribution.items():
            category_questions = self.db.get_questions(
                category=category,
                limit=int(count * percentage),
                difficulty=profile.learning_behavior["difficulty_preference"]
            )
            questions.extend(category_questions)
        
        return questions[:count]
```

### 4.3 Personalized Study Plan Generator

```python
class StudyPlanGenerator:
    """
    AI-generated individualized study plans
    """
    
    def generate_study_plan(
        self,
        user_id: str,
        exam_date: date,
        study_hours_per_week: int
    ) -> StudyPlan:
        """
        Create personalized study plan based on:
        - Days until exam
        - Current knowledge gaps
        - Available study time
        - Learning pace
        """
        profile = self.get_user_profile(user_id)
        days_available = (exam_date - date.today()).days
        
        # Calculate total study hours
        total_hours = (days_available / 7) * study_hours_per_week
        
        # Assess current readiness
        readiness = self.assess_readiness(profile)
        
        # Generate phases
        phases = []
        
        # Phase 1: Foundation (30% of time)
        # Cover all topics at least once
        foundation_weeks = int(days_available * 0.3 / 7)
        phases.append(StudyPhase(
            name="Foundation",
            duration_weeks=foundation_weeks,
            focus="Complete coverage of all AMC blueprint topics",
            daily_questions=study_hours_per_week * 2,  # ~2 min per question
            topics=self.get_all_blueprint_topics(),
            mode="adaptive"
        ))
        
        # Phase 2: Weakness Targeting (40% of time)
        # Focus on weak areas identified in Phase 1
        weakness_weeks = int(days_available * 0.4 / 7)
        weak_areas = self.identify_weak_areas(profile)
        phases.append(StudyPhase(
            name="Weakness Targeting",
            duration_weeks=weakness_weeks,
            focus=f"Intensive practice on weak areas: {', '.join(weak_areas[:5])}",
            daily_questions=study_hours_per_week * 3,  # More questions per hour
            topics=weak_areas,
            mode="weakness"
        ))
        
        # Phase 3: Exam Simulation (20% of time)
        # Timed practice exams
        exam_weeks = int(days_available * 0.2 / 7)
        phases.append(StudyPhase(
            name="Exam Simulation",
            duration_weeks=exam_weeks,
            focus="Timed practice exams under exam conditions",
            daily_questions=150,  # Full exam
            topics=[],
            mode="exam",
            exam_frequency="3x per week"
        ))
        
        # Phase 4: Final Review (10% of time)
        # High-yield topics and difficult questions
        review_weeks = int(days_available * 0.1 / 7)
        high_yield = self.get_high_yield_topics()
        phases.append(StudyPhase(
            name="Final Review",
            duration_weeks=review_weeks,
            focus="High-yield topics and marked difficult questions",
            daily_questions=study_hours_per_week * 2,
            topics=high_yield,
            mode="adaptive"
        ))
        
        return StudyPlan(
            user_id=user_id,
            exam_date=exam_date,
            total_hours=total_hours,
            phases=phases,
            milestones=self.generate_milestones(phases),
            readiness_target=80  # Target 80% readiness score
        )
    
    def adjust_plan(
        self,
        plan: StudyPlan,
        current_progress: Progress
    ) -> StudyPlan:
        """
        Dynamically adjust plan based on actual performance
        """
        # If ahead of schedule, add more difficult content
        if current_progress.ahead_of_schedule:
            plan.phases[1].topics.extend(self.get_advanced_topics())
        
        # If behind, focus on high-yield only
        if current_progress.behind_schedule:
            plan.phases[1].topics = self.get_high_yield_topics()
            plan.phases[2].exam_frequency = "2x per week"  # Reduce exam load
        
        # If specific topic struggles, add extra time
        for topic, accuracy in current_progress.topic_accuracy.items():
            if accuracy < 50:
                plan.add_extra_practice(topic, hours=2)
        
        return plan
```

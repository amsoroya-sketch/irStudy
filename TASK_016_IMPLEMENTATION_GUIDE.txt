- GET /progress/week/:week - Weekly progress

HOOK USAGE PATTERNS
===================

1. Fetch Data List:
   const { data, isLoading, error } = useMCQs({
     page: 1,
     limit: 20,
     difficulty: 'medium'
   });

2. Fetch Single Item:
   const { mcq, error, isLoading } = useMCQ(mcqId);

3. Submit Data:
   const { submitAnswer, isPending, isSuccess } = useMCQ(mcqId);
   submitAnswer.mutate({
     selectedAnswers: ['a', 'b'],
     timeSpent: 30
   });

4. Monitor Progress:
   const { data: progress } = useUserProgress({
     refetchInterval: 60 * 1000
   });

5. Handle Errors:
   if (error?.code === 'UNKNOWN_ERROR') {
     return <div>Network error</div>;
   }
